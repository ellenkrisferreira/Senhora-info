function getWeekRange(){

  const now=new Date();

  const day=now.getDay();

  const diff=day===0?-6:1-day;

  const start=new Date(now); start.setDate(now.getDate()+diff); start.setHours(0,0,0,0);

  const end=new Date(start); end.setDate(start.getDate()+6); end.setHours(23,59,59,999);

  return [start.getTime(),end.getTime()];

}

function getMonthRange(){

  const now=new Date();

  return [

    new Date(now.getFullYear(),now.getMonth(),1).getTime(),

    new Date(now.getFullYear(),now.getMonth()+1,0,23,59,59,999).getTime()

  ];

}

function contarConcluidasPeriodo(start,end){

  let c=0;

  for(const a in estado.aulasFeitas){

    const ts=estado.aulasFeitas[a];

    if(typeof ts==="number" && ts>=start && ts<=end) c++;

  }

  return c;

}

function exportarCSV(){

  let csv="Cargo,Curso,Módulo,Aula,Status,XP\n";

  DADOS.cargos.forEach(function(c){ c.cursos.forEach(function(cu){ cu.modulos.forEach(function(m){ m.aulas.forEach(function(a){

  });});});});

  const blob=new Blob([csv],{type:"text/csv;charset=utf-8;"});

  const url=URL.createObjectURL(blob);

  const a=document.createElement("a");

  a.href=url; a.download="senhora_info_"+new Date().toISOString().slice(0,10)+".csv"; a.click();

  URL.revokeObjectURL(url);

  mostrarToast("📄 CSV exportado!");

}

function importarBackup(ev){

  const f=ev.target.files[0];

  if(!f) return;

  const r=new FileReader();

  r.onload=function(e){

    try{

      const d=JSON.parse(e.target.result);

      if(d.xp!==undefined) estado.xp=d.xp;

      if(d.selos) estado.selos=d.selos;

      if(d.reliquias) estado.reliquias=d.reliquias;

      if(d.coroas) estado.coroas=d.coroas;

      if(d.metaSemanal) estado.metaSemanal=d.metaSemanal;

      if(d.metaMensal) estado.metaMensal=d.metaMensal;

      if(d.tema) estado.tema=d.tema;

      estado.aulasFeitas={};

      if(d.aulasFeitas){

        for(const k in d.aulasFeitas){

          const v=d.aulasFeitas[k];

          if(v===true) estado.aulasFeitas[k]=Date.now();

          else if(typeof v==="number") estado.aulasFeitas[k]=v;

        }

      }

      salvar();

      aplicarTema();

      render();

      renderColecao();

      mostrarToast("📥 Backup importado!");

    }catch(err){ alert("Arquivo inválido."); }

  };

  r.readAsText(f);

}

  }

}
